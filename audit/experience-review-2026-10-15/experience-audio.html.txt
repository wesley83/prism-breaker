import {Sound} from '/audio.js';import {TRACKS} from '/music.js';import {VICTORY_TRACKS} from '/victory-music.js';
document.querySelector('#render').onclick=async()=>{const result=[],status=document.querySelector('#status');try{const scenarios=[...TRACKS.map((track,level)=>({level,track})),...Object.entries(VICTORY_TRACKS).map(([victory,track])=>({level:0,victory,track}))];for(const scenario of scenarios)for(const cinematic of [false,true]){status.textContent='Rendering '+scenario.track.title+' '+(cinematic?'maximum cinematic mix':'default game mix');const duration=24,rate=44100,ctx=new OfflineAudioContext(2,duration*rate,rate),sound=new Sound({music:cinematic?1:.35,effects:cinematic?1:.7,muted:false});sound.offline=true;sound.setup(ctx);sound.active=true;sound.syncGain();const tick=60/scenario.track.bpm/4;for(let step=0;step*tick<duration-2;step++)sound.scheduleStep(scenario.level,step,.01+step*tick,cinematic,scenario.victory);if(cinematic){for(const type of ['paddle','wall','break','power','laser','life','clear','portal','champion','launch'])sound.event({type,kind:'explode'});}const buffer=await ctx.startRendering();let peak=0,sum=0,nonfinite=0,clipped=0;for(let ch=0;ch<2;ch++)for(const sample of buffer.getChannelData(ch)){if(!Number.isFinite(sample)){nonfinite++;continue;}peak=Math.max(peak,Math.abs(sample));sum+=sample*sample;if(Math.abs(sample)>=1)clipped++;}result.push({track:scenario.track.title,world:scenario.victory?null:scenario.level+1,victory:scenario.victory||null,scenario:cinematic?'maximum cinematic plus simultaneous effect burst':'default music only',seconds:duration,peak,rms:Math.sqrt(sum/(buffer.length*2)),nonfinite,clipped});document.querySelector('#result').textContent=JSON.stringify(result,null,2);}status.textContent='Complete: '+result.length+' renders';}catch(error){status.textContent='Failed: '+error.message;}};
</script></body></html>
